"use client"
import React, { useMemo, useState } from 'react'
import { motion, AnimatePresence } from 'motion/react'
import CategoryFilter from './CategoryFilter'
import PerfumeCard from './PerfumeCard'
import { Perfume, PerfumeCategory } from '../../types/perfume'

export default function PerfumeGrid({ perfumes, isLoading, lastUpdated }: { perfumes: Perfume[]; isLoading: boolean; lastUpdated: Date | null }){
  const [activeCategory, setActiveCategory] = useState<PerfumeCategory | 'All'>('All')
  const categories = useMemo(() => Array.from(new Set(perfumes.map(p=>p.category))) as PerfumeCategory[], [perfumes])
  const filtered = activeCategory === 'All' ? perfumes : perfumes.filter(p => p.category === activeCategory)

  if (isLoading) {
    return (
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
        {Array.from({length:6}).map((_,i)=> (
          <div key={i} className="shimmer rounded-2xl h-[360px]" />
        ))}
      </div>
    )
  }

  if (!filtered.length) {
    return <div className="text-[var(--gold)] italic text-center">No perfumes in this category.</div>
  }

  return (
    <div>
      <CategoryFilter categories={categories} active={activeCategory} onChange={(c) => setActiveCategory(c as any)} />
      <motion.div layout className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4 mt-4">
        <AnimatePresence>
          {filtered.map((p, idx) => (
            <motion.div key={p.id} initial={{ opacity:0, y:20 }} animate={{ opacity:1, y:0, transition: { delay: idx * 0.05 } }} exit={{ opacity:0, scale:0.95, transition:{ duration: 0.15 } }}>
              <PerfumeCard perfume={p} />
            </motion.div>
          ))}
        </AnimatePresence>
      </motion.div>
    </div>
  )
}

import React from 'react'
import { Perfume } from '../../types/perfume'

export default function PriceTicker({ perfumes }: { perfumes: Perfume[] }){
  const items = perfumes.map(p => {
    const lowest = p.pricing.reduce((a,b)=> Math.min(a,b.price), Number.MAX_SAFE_INTEGER)
    const arrow = p.trend === 'up' ? '▲' : p.trend === 'down' ? '▼' : '—'
    const color = p.trend === 'up' ? 'var(--price-up)' : p.trend === 'down' ? 'var(--price-down)' : 'var(--price-stable)'
    return { id: p.id, text: `${p.name} — ৳${lowest} ${arrow}`, color }
  })

  const loop = [...items, ...items]

  return (
    <div className="w-full bg-[var(--crimson-deep)] border-t border-b" style={{ borderColor: 'var(--border)', height: 40, display: 'flex', alignItems: 'center', overflow: 'hidden' }}>
      <div className="ticker px-6" style={{ display: 'flex', whiteSpace: 'nowrap' }}>
        {loop.map((it, idx) => (
          <span key={`${it.id}-${idx}`} style={{ marginRight: 24, fontFamily: 'var(--font-mono)', fontSize: 13, color: 'var(--gold-light)' }}>{it.text}</span>
        ))}
      </div>
    </div>
  )
}

import React, { useEffect, useRef, useState } from 'react'
import { motion, useReducedMotion } from 'motion/react'
import { formatBDT } from '../../lib/utils';
import { PriceTrend } from '../../types/perfume';

export default function AnimatedPrice({ price, trend, size = 'lg' }: { price: number; trend: PriceTrend; size?: 'sm' | 'lg' }){
  const reduced = useReducedMotion()
  const prev = useRef<number>(price)
  const [flash, setFlash] = useState<'up'|'down'|null>(null)

  useEffect(() => {
    if (price === prev.current) return
    const dir = price > prev.current ? 'up' : 'down'
    if (!reduced) {
      setFlash(dir)
      const t = setTimeout(() => setFlash(null), 800)
      return () => clearTimeout(t)
    }
    prev.current = price
  }, [price, reduced])

  const arrow = trend === 'up' ? '▲' : trend === 'down' ? '▼' : '—'
  const arrowColor = trend === 'up' ? 'var(--price-up)' : trend === 'down' ? 'var(--price-down)' : 'var(--price-stable)'

  return (
    <motion.div animate={{ backgroundColor: flash === 'up' ? 'rgba(74,222,128,0.08)' : flash === 'down' ? 'rgba(248,113,113,0.08)' : 'transparent' }} transition={{ duration: 0.6 }} className="inline-flex items-center gap-2">
      <span className="font-[var(--font-display)] text-[var(--gold-light)]" style={{ fontSize: size === 'lg' ? '1.25rem' : '0.95rem' }}>{formatBDT(price)}</span>
      <span aria-label={`Price trending ${trend}`} style={{ color: arrowColor, fontFamily: 'var(--font-mono)' }}>{arrow}</span>
    </motion.div>
  )
}

"use client"
import { useEffect, useRef, useState } from 'react'
import { Perfume } from '../types/perfume'
import { REFRESH_INTERVAL_MS } from '../lib/constants'

export function useLivePrices(intervalMs?: number){
  const [perfumes, setPerfumes] = useState<Perfume[]>([])
  const [isLoading, setIsLoading] = useState<boolean>(true)
  const [error, setError] = useState<string | null>(null)
  const [lastUpdated, setLastUpdated] = useState<Date | null>(null)
  const timer = useRef<number | null>(null)
  const ms = intervalMs ?? REFRESH_INTERVAL_MS

  async function fetchData(){
    try{
      setIsLoading(true)
      setError(null)
      const res = await fetch('/api/perfumes')
      if (!res.ok) throw new Error(`HTTP ${res.status}`)
      const data: Perfume[] = await res.json()
      setPerfumes(data)
      setLastUpdated(new Date())
    }catch(e:any){
      setError(e?.message ?? 'Unknown error')
    }finally{
      setIsLoading(false)
    }
  }

  useEffect(() => {
    fetchData()
    timer.current = window.setInterval(fetchData, ms)
    return () => { if (timer.current) window.clearInterval(timer.current) }
  }, [ms])

  return { perfumes, isLoading, error, lastUpdated }
}

export default useLivePrices

import { useEffect, useState } from 'react'

export function useScrollSpy(sectionIds: string[], offset = 80){
  const [activeSection, setActiveSection] = useState<string>(sectionIds[0] ?? '')

  useEffect(() => {
    if (typeof window === 'undefined') return
    const observers: IntersectionObserver[] = []
    const options = { threshold: 0.3, rootMargin: `-${offset}px 0px 0px 0px` }

    sectionIds.forEach(id => {
      const el = document.getElementById(id)
      if (!el) return
      const obs = new IntersectionObserver((entries) => {
        entries.forEach(entry => {
          if (entry.isIntersecting) setActiveSection(id)
        })
      }, options)
      obs.observe(el)
      observers.push(obs)
    })

    return () => observers.forEach(o => o.disconnect())
  }, [sectionIds.join(','), offset])

  return activeSection
}

export default useScrollSpy
